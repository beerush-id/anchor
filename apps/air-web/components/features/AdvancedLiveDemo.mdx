```tsx
import { effect, setup, Snippet } from '@airlib/react';
import { LIVE_KEYBOARD } from '@airlib/react/browser';

const LiveDemo = setup(() => {
  // Declarative shortcut handling without manual addEventListener
  effect.client(() => {
    if (LIVE_KEYBOARD.is('ctrl', '/') || LIVE_KEYBOARD.is('meta', '/')) {
      const input = document.querySelector<HTMLInputElement>('.air-search input');
      input?.focus();
    }
  });

  return (
    <div className="live-demo">
      <div className="live-demo-shortcuts">
        <div className="demo-box">
          <span>Focus Search</span>
          <kbd className="demo-kbd">Ctrl</kbd>
          <span>+</span>
          <kbd className="demo-kbd">/</kbd>
        </div>

        <div className="demo-box">
          <span>Pressed key</span>
          <Snippet>
            {() => <kbd className="demo-kbd">{LIVE_KEYBOARD.key || '—'}</kbd>}
          </Snippet>
        </div>
      </div>

      <p className="demo-hint">Press the shortcut anywhere on the page to focus the search bar.</p>
    </div>
  );
});
```
