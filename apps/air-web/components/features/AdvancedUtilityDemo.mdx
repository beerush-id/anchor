import { classx, mutable, setup, Show, sleep, Snippet, undoable } from '@airlib/react';
import { Badge } from '@airlib/react/mdx';

const UserCard = setup(() => {
  const user = mutable({ name: 'Aria Chen', handle: '@aria', likes: 128, liked: false });
  const demo = mutable({ pending: false, result: '' as '' | 'settled' | 'rolled back' });

  const toggleLike = async () => {
    demo.pending = true;

    // Optimistic mutation with undoable snapshot
    const [undo, settled] = undoable(() => {
      user.liked = !user.liked;
      user.likes += user.liked ? 1 : -1;
    });

    await sendLike().then(
      () => {
        settled();
        demo.result = 'settled';
      },
      () => {
        undo();
        demo.result = 'rolled back';
      }
    );

    demo.pending = false;
  };

  return (
    <div className="user-card-demo">
      <div className="demo-card">
        <div className="demo-avatar">AC</div>
        <div className="demo-info">
          <b>{user.name}</b>
          <small>{user.handle}</small>
        </div>

        <Snippet>
          {() => (
            <button
              type="button"
              className={classx('demo-like-btn', { active: user.liked })}
              disabled={demo.pending}
              onClick={toggleLike}
            >
              {user.liked ? '♥' : '♡'} {user.likes}
            </button>
          )}
        </Snippet>
      </div>

      <Show when={() => demo.result}>
        {(result) => <Badge variant={result === 'settled' ? 'tip' : 'destructive'}>{result.toUpperCase()}</Badge>}
      </Show>

      <p className="demo-hint">Optimistic updates apply instantly — if the request fails, undoable cleanly rolls them back.</p>
    </div>
  );
});

/**
 * Simulates a network request with latency and random rejection.
 */
const sendLike = () =>
  sleep(600).then(() => {
    if (Math.random() < 0.4) throw new Error('rejected');
  });
