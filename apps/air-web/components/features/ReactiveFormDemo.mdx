```tsx
import { Show, mutable, setup } from '@airlib/react';
import { createForm, EmailInput, Textarea, TextInput } from '@airlib/react-form';
import { z } from 'zod';

const schema = z.object({
  name: z.string().min(3, 'Name must be at least 3 characters'),
  email: z.string().email('Invalid email address'),
  bio: z.string().max(160, 'Bio must be at most 160 characters').optional(),
});

const UserForm = createForm(schema);

const ProfileForm = setup(() => {
  const payload = mutable('');

  return (
    <div className="flex w-full max-w-sm flex-col gap-3">
      <UserForm onSubmit={(_data, changes) => {
        payload.value = JSON.stringify(changes, null, 2);
      }}>
        <UserForm.Field name="name" label="Name">
          <TextInput placeholder="Enter name" />
        </UserForm.Field>

        <UserForm.Field name="email" label="Email">
          <EmailInput placeholder="name@company.com" />
        </UserForm.Field>

        <UserForm.Field name="bio" label="Bio">
          <Textarea placeholder="Tell us about yourself" />
        </UserForm.Field>

        <div className="flex gap-2">
          <UserForm.Reset>Reset</UserForm.Reset>
          <UserForm.Submit className="flex-1">Submit</UserForm.Submit>
        </div>
      </UserForm>

      <Show when={() => payload.value}>
        {(text) => (
          <>
            <div className="text-lg font-bold">Form Changes</div>
            <pre className="rounded-lg border border-border bg-surface p-3 font-mono text-xs text-on-surface-variant">
              {text}
            </pre>
          </>
        )}
      </Show>
    </div>
  );
});
```
