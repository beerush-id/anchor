```tsx
import { classx, cookies, For, setup } from '@airlib/react';

// The site's real theme cookie — the same reactive state the floating toggler renders
const ThemeCookie = setup(() => {
  const theme = cookies('app-theme', { mode: 'system' as ThemeMode });

  return (
    <div className="theme-selector">
      <div className="theme-selector-title">Select Theme</div>
      <div className="theme-selector-options">
        <For each={() => Object.entries(MODES)}>
          {([mode, label]) => (
            <button
              type="button"
              className={classx('demo-btn', { active: theme.mode === mode })}
              onClick={() => (theme.mode = mode)}
            >
              {label}
            </button>
          )}
        </For>
      </div>
      <p className="demo-hint">Pick a mode — the whole app follows, floating toggler included. Reload and it sticks.</p>
    </div>
  );
});

type ThemeMode = 'system' | 'light' | 'dark';

const MODES: { [K in ThemeMode]: string } = {
  system: 'System',
  light: 'Light',
  dark: 'Dark',
};
```
